+/* Extended bolus checks (WO-101). Separate from test_safety.c, which is protected evidence. */
+#include <assert.h>
+#include <math.h>
+#include <stdio.h>
+#include "pump.h"
+
+int main(void) {
+    pump_state s = {0};
+    assert(request_extended_bolus(&s, 24.0, 144, 120) == REJECT_LIMIT); /* not adult */
+    pump_set_adult(&s, 1);
+    assert(request_extended_bolus(&s, 24.0, 143, 120) == REJECT_LIMIT); /* > 10 U/h */
+    assert(request_extended_bolus(&s, 24.0, 144, 69) == REJECT_LOW_BG);
+    assert(request_extended_bolus(&s, 24.0, 144, 120) == OK);
+    assert(request_extended_bolus(&s, 1.0, 60, 120) == REJECT_LIMIT); /* one at a time */
+    assert(s.delivered_today_u == 0);                                    /* nothing until ticks */
+    for (int t = 0; t < 60; t++) assert(extended_bolus_tick(&s, 120) == OK);
+    assert(fabs(s.delivered_today_u - 10.0) < 1e-6);                     /* 10 U in first hour */
+    for (int t = 0; t < 84; t++) extended_bolus_tick(&s, 120);
+    assert(fabs(s.delivered_today_u - 24.0) < 1e-6 && s.ext_remaining_u == 0);
+
+    pump_state low = {0};
+    pump_set_adult(&low, 1);
+    assert(request_extended_bolus(&low, 6.0, 60, 120) == OK);
+    extended_bolus_tick(&low, 120);
+    assert(extended_bolus_tick(&low, 69) == REJECT_LOW_BG);              /* SR-3 per portion, cancels */
+    assert(low.ext_remaining_u == 0 && extended_bolus_tick(&low, 120) == OK && fabs(low.delivered_today_u - 0.1) < 1e-9);
+
+    pump_state day = {.delivered_today_u = 55.0};
+    pump_set_adult(&day, 1);
+    assert(request_extended_bolus(&day, 6.0, 60, 120) == REJECT_DAILY);  /* SR-2 up front */
+    puts("EXT PASS");
+    return 0;
+}

+void pump_set_adult(pump_state *s, int adult) { s->adult = adult != 0; }
+
+/* Large meals: total may exceed MAX_BOLUS_U, but delivery is spread so the rate stays <= MAX_BOLUS_U/h,
+ * and every per-minute portion goes through request_bolus (SR-1, SR-2, SR-3). */
+pump_status request_extended_bolus(pump_state *s, double units, int minutes, int bg_mg_dl) {
+    if (!s->adult || s->ext_remaining_u > 0) return REJECT_LIMIT;
+    if (units <= 0 || minutes <= 0 || minutes > MAX_EXT_MINUTES) return REJECT_LIMIT;
+    if (units * 60.0 > MAX_BOLUS_U * minutes + 1e-9) return REJECT_LIMIT; /* per-hour rate limit */
+    if (bg_mg_dl < MIN_BG_FOR_BOLUS) return REJECT_LOW_BG;
+    if (s->delivered_today_u + units > MAX_DAILY_U + 1e-9) return REJECT_DAILY;
+    s->ext_remaining_u = units;
+    s->ext_per_min_u = units / minutes;
+    return OK;
+}
+
+/* Low BG or daily cap cancels the rest of the extended bolus; it never resumes silently. */
+pump_status extended_bolus_tick(pump_state *s, int bg_mg_dl) {
+    if (s->ext_remaining_u <= 0) return OK;
+    double portion = s->ext_remaining_u < s->ext_per_min_u ? s->ext_remaining_u : s->ext_per_min_u;
+    pump_status r = request_bolus(s, portion, bg_mg_dl);
+    if (r != OK) { s->ext_remaining_u = 0; return r; }
+    s->ext_remaining_u -= portion;
+    if (s->ext_remaining_u < 1e-9) s->ext_remaining_u = 0;
+    return OK;
+}
